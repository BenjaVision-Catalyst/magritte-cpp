cmake_minimum_required(VERSION 3.20)
project(Magritte VERSION 0.4.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(magritte src/magritte.cpp src/collections.cpp src/portable.cpp src/editing.cpp src/self_description.cpp)
add_library(Magritte::magritte ALIAS magritte)
target_compile_features(magritte PUBLIC cxx_std_17)
target_include_directories(magritte PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE vendor)
set_target_properties(magritte PROPERTIES CXX_EXTENSIONS OFF WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MSVC)
  target_compile_options(magritte PRIVATE /W4)
else()
  target_compile_options(magritte PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS magritte EXPORT MagritteTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT MagritteTargets NAMESPACE Magritte:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Magritte)
configure_package_config_file(cmake/MagritteConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/MagritteConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Magritte)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MagritteConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MagritteConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/MagritteConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Magritte)
install(FILES LICENSE NOTICE.md vendor/nlohmann/LICENSE.MIT DESTINATION ${CMAKE_INSTALL_DATADIR}/Magritte/licenses)
include(CTest)
if(BUILD_TESTING)
  add_executable(magritte-tests tests/conformance.cpp)
  target_link_libraries(magritte-tests PRIVATE Magritte::magritte)
  target_include_directories(magritte-tests PRIVATE vendor)
  target_compile_definitions(magritte-tests PRIVATE FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/conformance")
  add_test(NAME magritte-conformance COMMAND magritte-tests)
  add_executable(magritte-portable-tests tests/portable.cpp)
  target_link_libraries(magritte-portable-tests PRIVATE Magritte::magritte)
  target_include_directories(magritte-portable-tests PRIVATE vendor)
  target_compile_definitions(magritte-portable-tests PRIVATE FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/conformance")
  add_test(NAME magritte-portable-conformance COMMAND magritte-portable-tests)
  add_executable(magritte-editing-tests tests/editing.cpp)
  target_link_libraries(magritte-editing-tests PRIVATE Magritte::magritte)
  target_include_directories(magritte-editing-tests PRIVATE vendor)
  target_compile_definitions(magritte-editing-tests PRIVATE FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/conformance")
  add_test(NAME magritte-editing-conformance COMMAND magritte-editing-tests)
endif()
