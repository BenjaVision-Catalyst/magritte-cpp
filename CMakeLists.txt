cmake_minimum_required(VERSION 3.20)
project(Magritte VERSION 0.15.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
add_library(magritte src/magritte.cpp src/collections.cpp src/portable.cpp src/editing.cpp src/self_description.cpp src/composition.cpp src/recursive.cpp src/reflection.cpp src/classes.cpp src/visitor.cpp src/options.cpp src/enumeration.cpp src/composite.cpp)
add_library(Magritte::magritte ALIAS magritte)
target_link_libraries(magritte PUBLIC Threads::Threads)
target_compile_features(magritte PUBLIC cxx_std_17)
target_include_directories(magritte PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE vendor)
set_target_properties(magritte PROPERTIES CXX_EXTENSIONS OFF WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MSVC)
  target_compile_options(magritte PRIVATE /W4)
else()
  target_compile_options(magritte PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS magritte EXPORT MagritteTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT MagritteTargets NAMESPACE Magritte:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Magritte)
configure_package_config_file(cmake/MagritteConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/MagritteConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Magritte)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MagritteConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MagritteConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/MagritteConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Magritte)
install(FILES LICENSE NOTICE.md vendor/nlohmann/LICENSE.MIT DESTINATION ${CMAKE_INSTALL_DATADIR}/Magritte/licenses)
include(CTest)
